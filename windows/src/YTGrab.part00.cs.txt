        void StartToolSetup()
        {
            addButton.Enabled = false;
            statusLabel.Text = "Preparing download engine...";
            Thread t = new Thread(new ThreadStart(EnsureToolsWorker));
            t.IsBackground = true;
            t.Start();
        }

        void EnsureToolsWorker()
        {
            try
            {
                Directory.CreateDirectory(toolsDir);
                ServicePointManager.SecurityProtocol = (SecurityProtocolType)3072;

                if (!File.Exists(ytDlpPath) || !HashMatches(ytDlpPath, YtDlpSha256))
                {
                    SetStatusSafe("Downloading yt-dlp engine...");
                    DownloadFileVerified(YtDlpUrl, ytDlpPath, YtDlpSha256);
                }

                if (!File.Exists(ffmpegPath))
                {
                    SetStatusSafe("Downloading Windows 8 compatible FFmpeg...");
                    string zipPath = Path.Combine(toolsDir, "ffmpeg.zip");
                    DownloadFileVerified(FfmpegUrl, zipPath, FfmpegSha256);
                    ExtractFfmpeg(zipPath);
                    TryDelete(zipPath);
                }

                if (!File.Exists(ffmpegPath)) throw new Exception("FFmpeg extraction failed.");

                toolsReady = true;
                manager.SetEngineReady(true);
                BeginInvoke((MethodInvoker)delegate
                {
                    addButton.Enabled = true;
                    statusLabel.Text = "Ready. Add downloads or open Active Downloads.";
                    engineLabel.Text = "Engine: yt-dlp 2026.08.19 | FFmpeg 7.0.2 | Windows 8 compatible";
                });
            }
            catch (Exception ex)
            {
                toolsReady = false;
                manager.SetEngineReady(false);
                BeginInvoke((MethodInvoker)delegate
                {
                    statusLabel.Text = "Engine setup failed: " + ex.Message;
                    engineLabel.Text = "Check your internet connection and reopen YTGrab.";
                    MessageBox.Show(this, "Could not prepare the download engine.\r\n\r\n" + ex.Message, "YTGrab", MessageBoxButtons.OK, MessageBoxIcon.Error);
                });
            }
        }

        void OpenAddWindow()
        {
            if (!toolsReady) return;
            AddDownloadForm form = new AddDownloadForm(ytDlpPath, toolsDir, defaultOutput);
            form.DownloadRequested += delegate(object sender, DownloadRequestEventArgs e)
            {
                defaultOutput = e.Folder;
                manager.AddAndStart(e.Url, e.VideoId, e.Title, e.Quality, e.Folder);
            };
            form.Show(this);
        }

            activeForm.FormClosed += delegate { activeForm = null; };
            activeForm.Show(this);
        }

        void ManagerJobsChanged()
        {
            if (IsDisposed) return;
            try
            {

                {
                    int count = manager.Count;
                    activeButton.Text = count > 0 ? "Active Downloads (" + count.ToString() + ")" : "Active Downloads";
                    activeSummaryLabel.Text = count == 0 ? "No active downloads." : count.ToString() + (count == 1 ? " active download" : " active downloads");
                });
            }
            catch { }
        }

        void DownloadFileVerified(string url, string destination, string expectedHash)
        {
            string temp = destination + ".download";
            TryDelete(temp);
            using (WebClient wc = new WebClient())
            {
                wc.Headers.Add("User-Agent", "YTGrab/1.3.0");
                wc.DownloadFile(url, temp);
            }
            if (!HashMatches(temp, expectedHash))
            {
                TryDelete(temp);
                throw new Exception("Downloaded engine file failed integrity verification.");
            }
            TryDelete(destination);
            File.Move(temp, destination);
        }

        bool HashMatches(string path, string expected)
        {
            try
            {
                using (SHA256 sha = SHA256.Create())
                using (FileStream fs = File.OpenRead(path))
                {
                    byte[] hash = sha.ComputeHash(fs);
                    StringBuilder sb = new StringBuilder(hash.Length * 2);
                    for (int i = 0; i < hash.Length; i++) sb.Append(hash[i].ToString("x2"));
                    return string.Equals(sb.ToString(), expected, StringComparison.OrdinalIgnoreCase);
                }
            }
            catch { return false; }
        }

        void ExtractFfmpeg(string zipPath)
        {
            using (FileStream fs = File.OpenRead(zipPath))
            using (ZipArchive archive = new ZipArchive(fs, ZipArchiveMode.Read))
            {
                foreach (ZipArchiveEntry entry in archive.Entries)
                {
                    string normalized = entry.FullName.Replace('\\', '/');
                    string destination = null;
                    if (normalized.EndsWith("/bin/ffmpeg.exe", StringComparison.OrdinalIgnoreCase)) destination = ffmpegPath;
                    if (normalized.EndsWith("/bin/ffprobe.exe", StringComparison.OrdinalIgnoreCase)) destination = ffprobePath;
                    if (destination == null) continue;
                    using (Stream input = entry.Open())
                    using (FileStream output = new FileStream(destination, FileMode.Create, FileAccess.Write, FileShare.None)) input.CopyTo(output);
                }
            }
        }

        void OpenFolder(string folder)
        {
            if (string.IsNullOrWhiteSpace(folder)) return;
            try
            {
                Directory.CreateDirectory(folder);
                Process.Start("explorer.exe", Quote(folder));
            }
            catch (Exception ex)
            {
                MessageBox.Show(this, ex.Message, "YTGrab", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        void SetStatusSafe(string text)
        {
            if (IsDisposed) return;
            try { BeginInvoke((MethodInvoker)delegate { statusLabel.Text = text; }); } catch { }
        }

        static string Quote(string s) { return "\"" + s.Replace("\"", "\\\"") + "\""; }
        static void TryDelete(string path) { try { if (File.Exists(path)) File.Delete(path); } catch { } }

        void OnFormClosing(object sender, FormClosingEventArgs e)
        {
            manager.PauseAllForShutdown();
        }
    }

        public DownloadRequestEventArgs(string url, string videoId, string title, string quality, string folder)
        {
            Url = url;
            VideoId = videoId;
            Title = title;
            Quality = quality;
            Folder = folder;
        }
    }
