        void AnalyzeUrl()
        {
            if (analyzing) return;
            string url = urlBox.Text.Trim();
            if (!IsYouTubeUrl(url))
            {
                MessageBox.Show(this, "Enter a valid YouTube URL.", "YTGrab", MessageBoxButtons.OK, MessageBoxIcon.Information);
                return;
            }

            analyzing = true;
            currentVideoId = "";
            analyzeButton.Enabled = false;
            downloadButton.Enabled = false;
            qualityBox.Enabled = false;
            titleLabel.Text = "Analyzing video...";
            statusLabel.Text = "Reading available formats...";

            Thread t = new Thread(new ThreadStart(delegate
            {
                try
                {
                    string info = RunAndCapture(ytDlpPath, "--no-warnings --no-playlist --print " + Quote("%(id)s") + " --print " + Quote("%(title)s") + " " + Quote(url));
                    string formats = RunAndCapture(ytDlpPath, "--no-warnings --no-playlist -F " + Quote(url));
                    List<string> infoLines = NonEmptyLines(info);
                    string videoId = infoLines.Count > 0 ? infoLines[0] : "";
                    string videoTitle = infoLines.Count > 1 ? infoLines[1] : "Video detected";
                    List<int> heights = ParseHeights(formats);

                    if (IsDisposed) return;
                    try
                    {
                        BeginInvoke((MethodInvoker)delegate
                        {
                            currentVideoId = videoId;
                            titleLabel.Text = videoTitle;
                            qualityBox.Items.Clear();
                            qualityBox.Items.Add("Best available");
                            foreach (int h in heights)
                            {
                                if (h >= 2160) qualityBox.Items.Add(h.ToString() + "p (4K)");
                                else if (h == 1440) qualityBox.Items.Add("1440p (2K)");
                                else qualityBox.Items.Add(h.ToString() + "p");
                            }
                            qualityBox.Items.Add("Audio only (MP3)");
                            qualityBox.SelectedIndex = 0;
                            qualityBox.Enabled = true;
                            downloadButton.Enabled = true;
                            analyzeButton.Enabled = true;
                            statusLabel.Text = "Ready to add to downloads.";
                            analyzing = false;
                        });
                    }
                    catch { }
                }
                catch (Exception ex)
                {
                    try
                    {
                        BeginInvoke((MethodInvoker)delegate
                        {
                            analyzeButton.Enabled = true;
                            titleLabel.Text = "Could not analyze this video.";
                            statusLabel.Text = ex.Message;
                            analyzing = false;
                            MessageBox.Show(this, ex.Message, "YTGrab", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        });
                    }
                    catch { }
                }
            }));
            t.IsBackground = true;
            t.Start();
        }

        void SubmitDownload()
        {
            if (qualityBox.SelectedItem == null) return;
            string url = urlBox.Text.Trim();
            string folder = outputBox.Text.Trim();
            if (!IsYouTubeUrl(url) || string.IsNullOrWhiteSpace(folder)) return;
            try { Directory.CreateDirectory(folder); }
            catch (Exception ex)
            {
                MessageBox.Show(this, "Cannot use that output folder.\r\n\r\n" + ex.Message, "YTGrab", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            EventHandler<DownloadRequestEventArgs> handler = DownloadRequested;
            if (handler != null) handler(this, new DownloadRequestEventArgs(url, currentVideoId, titleLabel.Text, qualityBox.SelectedItem.ToString(), folder));

            currentVideoId = "";
            urlBox.Clear();
            qualityBox.Items.Clear();
            qualityBox.Enabled = false;
            downloadButton.Enabled = false;
            analyzeButton.Enabled = true;
            titleLabel.Text = "Download started in the background. Paste another YouTube link.";
            statusLabel.Text = "Added. View progress from Active Downloads when you want.";
            urlBox.Focus();
        }

        static List<int> ParseHeights(string text)
        {
            HashSet<int> heights = new HashSet<int>();
            MatchCollection a = Regex.Matches(text, @"\b\d{2,5}x(\d{3,4})\b", RegexOptions.IgnoreCase);
            foreach (Match m in a)
            {
                int h;
                if (int.TryParse(m.Groups[1].Value, out h) && h >= 144 && h <= 4320) heights.Add(h);
            }
            MatchCollection b = Regex.Matches(text, @"\b(\d{3,4})p\b", RegexOptions.IgnoreCase);
            foreach (Match m in b)
            {
                int h;
                if (int.TryParse(m.Groups[1].Value, out h) && h >= 144 && h <= 4320) heights.Add(h);
            }
            List<int> result = heights.ToList();
            result.Sort(delegate(int x, int y) { return y.CompareTo(x); });
            return result;
        }

        static string RunAndCapture(string fileName, string arguments)
        {
            ProcessStartInfo psi = new ProcessStartInfo();
            psi.FileName = fileName;
            psi.Arguments = arguments;
            psi.UseShellExecute = false;
            psi.CreateNoWindow = true;
            psi.RedirectStandardOutput = true;
            psi.RedirectStandardError = true;
            psi.StandardOutputEncoding = Encoding.UTF8;
            psi.StandardErrorEncoding = Encoding.UTF8;
            using (Process p = Process.Start(psi))
            {
                string output = p.StandardOutput.ReadToEnd();
                string error = p.StandardError.ReadToEnd();
                p.WaitForExit();
                if (p.ExitCode != 0)
                {
                    string message = LastUsefulLine(error);
                    if (string.IsNullOrWhiteSpace(message)) message = "yt-dlp returned an error.";
                    throw new Exception(message);
                }
                return output;
            }
        }

        static bool IsYouTubeUrl(string text)
        {
            Uri uri;
            if (!Uri.TryCreate(text, UriKind.Absolute, out uri)) return false;
            if (uri.Scheme != Uri.UriSchemeHttp && uri.Scheme != Uri.UriSchemeHttps) return false;
            string host = uri.Host.ToLowerInvariant();
            return host == "youtu.be" || host == "youtube.com" || host.EndsWith(".youtube.com") || host == "youtube-nocookie.com" || host.EndsWith(".youtube-nocookie.com");
        }

        static string Quote(string s) { return "\"" + s.Replace("\"", "\\\"") + "\""; }

        static List<string> NonEmptyLines(string text)
        {
            List<string> result = new List<string>();
            if (text == null) return result;
            string[] lines = text.Replace("\r", "").Split('\n');
            for (int i = 0; i < lines.Length; i++) if (!string.IsNullOrWhiteSpace(lines[i])) result.Add(lines[i].Trim());
            return result;
        }

        static string LastUsefulLine(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return "";
            string[] lines = text.Replace("\r", "").Split('\n');
            for (int i = lines.Length - 1; i >= 0; i--)
            {
                string line = lines[i].Trim();
                if (!string.IsNullOrWhiteSpace(line)) return line;
            }
            return "";
        }
    }

    public sealed class DownloadManager
    {
        readonly object sync = new object();
        readonly List<DownloadJob> jobs = new List<DownloadJob>();
        readonly string ytDlpPath;
        readonly string toolsDir;
        readonly string statePath;
        volatile bool engineReady;

        public event Action JobsChanged;

        public DownloadManager(string ytDlpPath, string toolsDir)
        {
            this.ytDlpPath = ytDlpPath;
            this.toolsDir = toolsDir;
            string stateDir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "YTGrab");
            statePath = Path.Combine(stateDir, "jobs.dat");
        }

        public int Count
        {
            get { lock (sync) { return jobs.Count; } }
        }

        public bool EngineReady { get { return engineReady; } }

        public void SetEngineReady(bool value)
        {
            engineReady = value;
            FireJobsChanged();
        }

        public List<DownloadJob> Snapshot()
        {
            lock (sync) { return new List<DownloadJob>(jobs); }
        }

        public void LoadSavedJobs()
        {
            List<DownloadJobModel> models = JobStore.Load(statePath);
            lock (sync)
            {
                jobs.Clear();
                foreach (DownloadJobModel model in models)
                {
                    if (string.Equals(model.State, "Completed", StringComparison.OrdinalIgnoreCase)) continue;
                    if (string.Equals(model.State, "Downloading", StringComparison.OrdinalIgnoreCase) || string.Equals(model.State, "Pausing", StringComparison.OrdinalIgnoreCase))
                    {
                        model.State = "Paused";
                        model.Status = "Interrupted by shutdown or restart. Click Resume to continue.";
                    }
                    DownloadJob job = CreateJob(model);
                    jobs.Add(job);
                }
            }
            SaveState();
            FireJobsChanged();
        }

        public DownloadJob AddAndStart(string url, string videoId, string title, string quality, string folder)
        {
            DownloadJobModel model = new DownloadJobModel();
            model.JobId = Guid.NewGuid().ToString("N");
            model.Url = url;
            model.VideoId = videoId == null ? "" : videoId.Trim();
            model.Title = string.IsNullOrWhiteSpace(title) ? "YouTube video" : title;
            model.Quality = quality;
            model.Folder = folder;
            model.Progress = 0;
            model.State = "Paused";
            model.Status = "Queued...";
            DownloadJob job = CreateJob(model);
            lock (sync) { jobs.Add(job); }
            SaveState();
            FireJobsChanged();
            job.Resume();
            return job;
        }

        DownloadJob CreateJob(DownloadJobModel model)
        {
            DownloadJob job = new DownloadJob(this, model, ytDlpPath, toolsDir);
            job.Changed += delegate { SaveState(); };
            job.Completed += delegate { RemoveCompleted(job); };
            return job;
        }

        void RemoveCompleted(DownloadJob job)
        {
            lock (sync) { jobs.Remove(job); }
            SaveState();
            FireJobsChanged();
        }

        {
